 //Accessibility additions
 (function () {
 // -----------------------------
 // Configuration
 // -----------------------------
 const REPORT_SELECTOR = '#reportContainer';
 const INNER_TABLE_SELECTOR = '#reportContainer table.jrPage, #reportContainer table[id^="JR_PAGE_ANCHOR"]';
 // -----------------------------
 // Utility
 // -----------------------------
 function slugify(text) {
 return String(text || '')
 .toLowerCase()
 .trim()
 .replace(/[^a-z0-9]+/g, '-')
 .replace(/^-+|-+$/g, '');
 }
 function getInnerTable() {
 return document.querySelector(INNER_TABLE_SELECTOR);
 }
 function getAllRows() {
 const table = getInnerTable();
 return table ? Array.from(table.querySelectorAll('tr')) : [];
 }
 function getCellText(cell) {
 return (cell ? cell.innerText || '' : '').replace(/\s+/g, ' ').trim();
 }
 function hasMeaningfulText(cell) {
 return getCellText(cell).length > 0;
 }
 function isSpacerCell(cell) {
 if (!cell) return true;
 return !hasMeaningfulText(cell) && !cell.querySelector('a, button, input, select, textarea');
 }
 function isMeaningfulCell(cell) {
 return !isSpacerCell(cell);
 }
 function isGreenBarCell(cell) {
 return cell && window.getComputedStyle(cell).backgroundColor === 'rgb(40, 128, 124)';
 }
 function isGreenBarRow(row) {
 return Array.from(row.children).some(isGreenBarCell);
 }
 function getMeaningfulCells(row) {
 if (!row) return [];
 return Array.from(row.children).filter(isMeaningfulCell);
 }
 function isHeaderRow(row) {
 const text = getCellText(row).toLowerCase();
 return (
 text.includes('status') &&
 text.includes('party') &&
 text.includes('candidate name') &&
 text.includes('filed on') &&
 text.includes('filing method')
 );
 }
 function isLikelyDataRow(row) {
 if (!row || isGreenBarRow(row) || isHeaderRow(row)) return false;
 const meaningfulCells = getMeaningfulCells(row);
 if (meaningfulCells.length {
 if (!exclude.has(attr.name)) {
 toEl.setAttribute(attr.name, attr.value);
 }
 });
 }
 function replaceTdWithTh(td, idBase) {
 const th = document.createElement('th');
 copyAttributes(td, th, []);
 th.innerHTML = td.innerHTML;
 th.setAttribute('scope', 'col');
 const text = getCellText(th);
 if (text) {
 th.id = idBase || ('col-' + slugify(text));
 th.setAttribute('abbr', text);
 }
 td.parentNode.replaceChild(th, td);
 return th;
 }
 function findCurrentSectionHeadingId(row) {
 let prev = row.previousElementSibling;
 while (prev) {
 if (isGreenBarRow(prev)) {
 const headingCell = Array.from(prev.children).find(isGreenBarCell);
 if (headingCell) return headingCell.id || null;
 }
 prev = prev.previousElementSibling;
 }
 return null;
 }
 function clearCellTabStops(table) {
 if (!table) return;
 table.querySelectorAll('td[tabindex="0"], th[tabindex="0"]').forEach(el => {
 el.setAttribute('tabindex', '-1');
 });
 }
 function focusRow(row) {
 if (!row) return;

 }
 function focusCell(cell, table) {
 if (!cell) return;

 }
 function getInteractiveCellSet(row) {
 if (!row) return [];

 });
 }
 function findAdjacentDataRow(startRow, direction) {
 let row = startRow;
 while (row) {
 row = direction === 'up' ? row.previousElementSibling : row.nextElementSibling;
 if (!row) return null;
 if (isLikelyDataRow(row)) return row;
 }
 return null;
 }

 table.querySelectorAll('td').forEach(cell => {

 th.setAttribute('scope', 'col');
 }
 th.removeAttribute('tabindex');
 th.setAttribute('role', 'columnheader');
 for (let i = 0; i {
 if (isSpacerCell(cell)) {
 cell.setAttribute('role', 'presentation');
 cell.setAttribute('aria-hidden', 'true');
 }
 });
 // Apply headers to data cells based on visual column position
 rows.forEach(row => {

 let visualIndex = 0;
 Array.from(row.children).forEach(cell => {
 const colspan = getColSpan(cell);
 if (isSpacerCell(cell)) {
 cell.setAttribute('role', 'presentation');
 cell.setAttribute('aria-hidden', 'true');
 visualIndex += colspan;
 return;
 }
 const headerId = colHeaders[visualIndex];
 if (headerId) {
 cell.setAttribute('headers', headerId);
 }
 visualIndex += colspan;
 });
 });
 }

 function enhanceRows() {

 const rows = Array.from(table.querySelectorAll('tr'));

 rows.forEach(row => {

 row.setAttribute('role', 'row');

 dataRowCount += 1;
 row.setAttribute('tabindex', '0');
 const sectionHeadingId = findCurrentSectionHeadingId(row);
 if (sectionHeadingId) {
 row.setAttribute('aria-labelledby', sectionHeadingId);
 }
 row.setAttribute(
 'aria-label',
 'Candidate row. Press Enter or Right Arrow to move into row cells.'
 );
 } else {
 row.removeAttribute('tabindex');
 }
 });
 }

 // -----------------------------
 // 7. Keyboard navigation
 // -----------------------------
 function enableKeyboardNavigation() {
 const table = getInnerTable();
 if (!table) return;
 document.addEventListener('keydown', function (e) {
 if (e.altKey || e.ctrlKey || e.metaKey) return;
 const active = document.activeElement;
 if (!active) return;
 const isRow = active.tagName === 'TR';
 const isCell = active.tagName === 'TD' || active.tagName === 'TH';
 // Row -> first cell
 if (isRow) {
 if (e.key === 'Enter' || e.key === 'ArrowRight') {
 const cells = getInteractiveCellSet(active).filter(cell => cell.tagName !== 'TH' || !isHeaderRow(active));
 const firstDataCell = cells.find(cell => cell.tagName === 'TD') || cells[0];
 if (firstDataCell) {
 e.preventDefault();
 focusCell(firstDataCell, table);
 }
 }
 return;
 }
 if (!isCell) return;
 if (!table.contains(active)) return;
 // Leave section heading behavior alone
 if (active.getAttribute('role') === 'heading') return;
 const row = active.parentElement;
 if (!row) return;
 const cells = getInteractiveCellSet(row);
 const currentIndex = cells.indexOf(active);
 if (currentIndex === -1) return;
 let target = null;
 switch (e.key) {
 case 'ArrowRight':
 target = cells[currentIndex + 1] || null;
 break;
 case 'ArrowLeft':
 target = cells[currentIndex - 1] || null;
 break;
 case 'ArrowDown': {
 const nextRow = findAdjacentDataRow(row, 'down');
 if (nextRow) {
 const nextCells = getInteractiveCellSet(nextRow);
 target = nextCells[currentIndex] || nextCells[nextCells.length - 1] || null;
 }
 break;
 }
 case 'ArrowUp': {
 const prevRow = findAdjacentDataRow(row, 'up');
 if (prevRow) {
 const prevCells = getInteractiveCellSet(prevRow);
 target = prevCells[currentIndex] || prevCells[prevCells.length - 1] || null;
 }
 break;
 }
 case 'Escape':
 e.preventDefault();
 clearCellTabStops(table);
 focusRow(row);
 return;
 default:
 return;
 }
 if (target) {
 e.preventDefault();
 focusCell(target, table);
 }
 });
 }
